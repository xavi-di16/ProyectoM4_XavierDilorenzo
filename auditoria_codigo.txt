# React + TypeScript + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the Oxlint configuration

If you are developing a production application, we recommend enabling type-aware lint rules by installing `oxlint-tsgolint` and editing `.oxlintrc.json`:

```json
{
  "$schema": "./node_modules/oxlint/configuration_schema.json",
  "plugins": ["react", "typescript", "oxc"],
  "options": {
    "typeAware": true
  },
  "rules": {
    "react/rules-of-hooks": "error",
    "react/only-export-components": ["warn", { "allowConstantExport": true }]
  }
}
```

See the [Oxlint rules documentation](https://oxc.rs/docs/guide/usage/linter/rules) for the full list of rules and categories.

import type {
    VercelRequest,
    VercelResponse,
} from '@vercel/node';

import { getAdminServices } from '../server/firebaseAdmin.js';
import { sendSummaryEmail } from '../server/email.js';
import { parseTask } from '../src/utils/taskParser.js';

class RateLimitError extends Error { }

export default async function handler(
    req: VercelRequest,
    res: VercelResponse,
) {
    res.setHeader('Cache-Control', 'no-store');

    if (req.method !== 'POST') {
        res.setHeader('Allow', 'POST');

        return res.status(405).json({
            message: 'Usá POST para solicitar el resumen.',
        });
    }

    const contentType = req.headers['content-type']
        ?.split(';')[0]
        ?.trim()
        .toLowerCase();

    if (contentType !== 'application/json') {
        return res.status(415).json({
            message: 'La solicitud debe usar application/json.',
        });
    }

    const authorization = req.headers.authorization;

    if (
        !authorization?.startsWith('Bearer ') ||
        !authorization.slice(7).trim()
    ) {
        return res.status(401).json({
            message: 'Necesitás iniciar sesión.',
        });
    }

    let body: unknown = req.body;

    if (typeof body === 'string') {
        try {
            body = JSON.parse(body);
        } catch {
            return res.status(400).json({
                message: 'El JSON recibido es inválido.',
            });
        }
    }

    if (
        !body ||
        typeof body !== 'object' ||
        Array.isArray(body) ||
        Object.keys(body).length !== 0
    ) {
        return res.status(400).json({
            message: 'El cuerpo de la solicitud debe ser {}.',
        });
    }

    try {
        const { auth, db } = getAdminServices();

        const token = authorization.slice(7).trim();

        const decoded = await auth
            .verifyIdToken(token)
            .catch(() => null);

        if (!decoded) {
            return res.status(401).json({
                message: 'La sesión es inválida o está vencida.',
            });
        }

        if (!decoded.email || !decoded.email_verified) {
            return res.status(403).json({
                message:
                    'Verificá tu correo en Firebase antes de enviar el resumen.',
            });
        }

        // Admin SDK omite las reglas de Firestore.
        // Este filtro por el UID verificado es obligatorio.
        const snapshot = await db
            .collection('tasks')
            .where('userId', '==', decoded.uid)
            .get();

        const tasks = snapshot.docs.map(document =>
            parseTask(document.id, document.data()),
        );

        if (tasks.some(task => task.userId !== decoded.uid)) {
            throw new Error('Se encontró un propietario inválido.');
        }

        tasks.sort((a, b) => b.createdAt - a.createdAt);

        const limitReference = db
            .collection('_emailLimits')
            .doc(decoded.uid);

        await db.runTransaction(async transaction => {
            const previous = await transaction.get(limitReference);

            const lastAttempt: unknown =
                previous.data()?.lastAttempt;

            const now = Date.now();

            if (
                typeof lastAttempt === 'number' &&
                now - lastAttempt < 60_000
            ) {
                throw new RateLimitError();
            }

            transaction.set(limitReference, {
                lastAttempt: now,
            });
        });

        const messageId = await sendSummaryEmail(
            decoded.email,
            tasks,
        );

        return res.status(200).json({
            message:
                'AWS SES aceptó el resumen. Revisá tu bandeja de entrada y spam.',
            messageId,
        });
    } catch (error) {
        if (error instanceof RateLimitError) {
            res.setHeader('Retry-After', '60');

            return res.status(429).json({
                message:
                    'Esperá un minuto entre intentos de envío.',
            });
        }

        console.error('Error en send-summary:', {
            name: error instanceof Error
                ? error.name
                : 'UnknownError',
            message: error instanceof Error
                ? error.message
                : 'Error desconocido',
        });

        if (
            error instanceof Error &&
            error.name === 'MessageRejected'
        ) {
            return res.status(422).json({
                message:
                    'SES rechazó el envío. Revisá las identidades verificadas y la región de AWS.',
            });
        }

        return res.status(500).json({
            message:
                'No se pudo enviar el resumen. Revisá la terminal del servidor.',
        });
    }
}

import {
    SESv2Client,
    SendEmailCommand,
} from '@aws-sdk/client-sesv2';

import type { Task } from '../src/types/task.types.js';

export function escapeHtml(value: string): string {
    const replacements: Record<string, string> = {
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#39;',
    };

    return value.replace(
        /[&<>"']/g,
        character => replacements[character] ?? character,
    );
}

export function buildTaskSummary(tasks: Task[]) {
    const completed = tasks.filter(task => task.completed).length;
    const pending = tasks.length - completed;

    const summary =
        `Total: ${tasks.length} | ` +
        `Completadas: ${completed} | ` +
        `Pendientes: ${pending}`;

    const lines = tasks.map(task => {
        const state = task.completed
            ? '[Completada]'
            : '[Pendiente]';

        return (
            `${state} ${task.title}` +
            (task.description ? ` — ${task.description}` : '')
        );
    });

    return {
        subject: `MateCode: resumen de tareas (${pending} pendientes)`,

        text:
            `${summary}\n\n` +
            (lines.join('\n') || 'No tenés tareas cargadas.'),

        html:
            '<h1>Tu resumen de MateCode</h1>' +
            `<p>${summary}</p>` +
            '<ul>' +
            lines
                .map(line => `<li>${escapeHtml(line)}</li>`)
                .join('') +
            '</ul>' +
            (tasks.length
                ? ''
                : '<p>No tenés tareas cargadas.</p>'),
    };
}

export async function sendSummaryEmail(
    recipient: string,
    tasks: Task[],
): Promise<string | undefined> {
    const region = process.env.AWS_REGION;
    const accessKeyId = process.env.AWS_ACCESS_KEY_ID;
    const secretAccessKey = process.env.AWS_SECRET_ACCESS_KEY;
    const fromEmail = process.env.AWS_SES_FROM_EMAIL;

    if (
        !region ||
        !accessKeyId ||
        !secretAccessKey ||
        !fromEmail
    ) {
        throw new Error(
            'Faltan variables de entorno de AWS SES.',
        );
    }

    const client = new SESv2Client({
        region,
        credentials: {
            accessKeyId,
            secretAccessKey,
        },
    });

    const content = buildTaskSummary(tasks);

    const result = await client.send(
        new SendEmailCommand({
            FromEmailAddress: fromEmail,
            Destination: {
                ToAddresses: [recipient],
            },
            Content: {
                Simple: {
                    Subject: {
                        Data: content.subject,
                        Charset: 'UTF-8',
                    },
                    Body: {
                        Text: {
                            Data: content.text,
                            Charset: 'UTF-8',
                        },
                        Html: {
                            Data: content.html,
                            Charset: 'UTF-8',
                        },
                    },
                },
            },
        }),
    );

    return result.MessageId;
}

import {
    cert,
    getApps,
    initializeApp,
} from 'firebase-admin/app';

import { getAuth } from 'firebase-admin/auth';
import { getFirestore } from 'firebase-admin/firestore';

export function getAdminServices() {
    const projectId = process.env.FIREBASE_PROJECT_ID;
    const clientEmail = process.env.FIREBASE_CLIENT_EMAIL;
    const privateKey = process.env.FIREBASE_PRIVATE_KEY
        ?.replace(/\\n/g, '\n');

    if (!projectId || !clientEmail || !privateKey) {
        throw new Error(
            'Faltan variables de entorno de Firebase Admin.',
        );
    }

    const app =
        getApps().find(item => item.name === 'matecode-server') ??
        initializeApp(
            {
                credential: cert({
                    projectId,
                    clientEmail,
                    privateKey,
                }),
                projectId,
            },
            'matecode-server',
        );

    return {
        auth: getAuth(app),
        db: getFirestore(app),
    };
}

export interface AuthFormErrors {
    email?: string;
    password?: string;
    confirmPassword?: string;
}

export function validateAuth(
    values: AuthFormValues,
    isRegister: boolean,
    confirmPassword: string,
): AuthFormErrors {
    const errors: AuthFormErrors = {};

    const email = values.email.trim();
    const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

    if (!emailPattern.test(email)) {
        errors.email = 'Ingresá un correo electrónico válido.';
    }

    if (!values.password) {
        errors.password = 'Ingresá tu contraseña.';
    } else if (isRegister && values.password.length < 6) {
        errors.password =
            'La contraseña debe tener al menos 6 caracteres.';
    }

    if (isRegister && confirmPassword !== values.password) {
        errors.confirmPassword = 'Las contraseñas no coinciden.';
    }

    return errors;
}

import type { Task } from '../types/task.types';
import { validateTask } from './validateTask.js';

export function parseTask(
    id: string,
    raw: unknown,
): Task {
    if (
        !raw ||
        typeof raw !== 'object' ||
        Array.isArray(raw)
    ) {
        throw new Error('Hay una tarea con formato inválido.');
    }

    const data = raw as Record<string, unknown>;

    const {
        userId,
        title,
        description,
        completed,
        createdAt,
        updatedAt,
    } = data;

    if (
        typeof userId !== 'string' ||
        !userId ||
        typeof title !== 'string' ||
        typeof description !== 'string' ||
        typeof completed !== 'boolean' ||
        typeof createdAt !== 'number' ||
        !Number.isSafeInteger(createdAt) ||
        createdAt < 0 ||
        typeof updatedAt !== 'number' ||
        !Number.isSafeInteger(updatedAt) ||
        updatedAt < createdAt
    ) {
        throw new Error(
            'Hay una tarea con datos incompatibles en Firestore.',
        );
    }

    const errors = validateTask({ title, description });

    if (errors.title || errors.description) {
        throw new Error(
            'Hay una tarea que no cumple las validaciones.',
        );
    }

    return {
        id,
        userId,
        title,
        description,
        completed,
        createdAt,
        updatedAt,
    };
}

const COLLECTION_NAME = 'tasks';

function getValidatedValues(
    values: TaskFormValues,
): TaskFormValues {
    const errors = validateTask(values);

    if (errors.title || errors.description) {
        throw new Error(
            errors.title ??
            errors.description ??
            'Los datos de la tarea son inválidos.',
        );
    }

    return {
        title: values.title.trim(),
        description: values.description.trim(),
    };
}

export function subscribeToUserTasks(
    userId: string,
    onData: (tasks: Task[]) => void,
    onError: (error: Error) => void,
): Unsubscribe {
    const userTasksQuery = query(
        collection(db, COLLECTION_NAME),
        where('userId', '==', userId),
    );

    return onSnapshot(
        userTasksQuery,
        snapshot => {
            try {
                const tasks = snapshot.docs.map(document =>
                    parseTask(document.id, document.data()),
                );

                if (tasks.some(task => task.userId !== userId)) {
                    throw new Error(
                        'Se recibió una tarea con un propietario inválido.',
                    );
                }

                tasks.sort(
                    (a, b) =>
                        b.createdAt - a.createdAt ||
                        a.id.localeCompare(b.id),
                );

                onData(tasks);
            } catch (error) {
                onError(
                    error instanceof Error
                        ? error
                        : new Error('No se pudieron leer las tareas.'),
                );
            }
        },
        onError,
    );
}

export async function createTask(
    userId: string,
    values: TaskFormValues,
): Promise<void> {
    if (!userId) {
        throw new Error('Necesitás iniciar sesión.');
    }

    const validatedValues = getValidatedValues(values);
    const now = Date.now();

    await addDoc(collection(db, COLLECTION_NAME), {
        userId,
        ...validatedValues,
        completed: false,

        updatedAt: now,
    });
}

    useCallback,
    useEffect,
    useState,
} from 'react';

import {
    createTask,
    deleteTask,
    setTaskCompleted,
    subscribeToUserTasks,
    updateTask,
} from '../features/tasks/taskService';

import type {
    Task,
    TaskFilter,
    TaskFormValues,
} from '../types/task.types';

import { getTaskErrorMessage } from '../utils/taskErrors';

interface TasksState {
    owner: string;
    tasks: Task[];
    loading: boolean;
    error: string | null;
}

export function useTasks(userId: string) {
    const [state, setState] = useState<TasksState>({
        owner: userId,
        tasks: [],
        loading: true,
        error: null,
    });

    const [filter, setFilter] = useState<TaskFilter>('all');

    useEffect(() => {
        let active = true;

        const unsubscribe = subscribeToUserTasks(
            userId,
            tasks => {
                if (!active) return;

                setState({
                    owner: userId,
                    tasks,
                    loading: false,
                    error: null,
                });
            },
            failure => {
                if (!active) return;

                setState({
                    owner: userId,
                    tasks: [],
                    loading: false,
                    error: getTaskErrorMessage(failure),
                });
            },
        );

        return () => {
            active = false;
            unsubscribe();
        };
    }, [userId]);

    const current: TasksState =
        state.owner === userId
            ? state
            : {
                owner: userId,
                tasks: [],
                loading: true,
                error: null,
            };

    const addTask = useCallback(
        (values: TaskFormValues) =>
            createTask(userId, values),
        [userId],
    );

    const editTask = useCallback(
        (taskId: string, values: TaskFormValues) =>
            updateTask(taskId, values),
        [],
    );

    const toggleTask = useCallback(
        (task: Task) =>
            setTaskCompleted(task.id, !task.completed),
        [],
    );

    const removeTask = useCallback(
        (taskId: string) => deleteTask(taskId),
        [],
    );

    const { tasks, loading, error } = current;

    const filteredTasks = tasks.filter(task => {
        if (filter === 'pending') return !task.completed;
        if (filter === 'completed') return task.completed;

        return true;
    });

    const counts = {
        all: tasks.length,
        pending: tasks.filter(task => !task.completed).length,
        completed: tasks.filter(task => task.completed).length,
    };

    return {
        tasks,
        filteredTasks,
        counts,
        loading,
        error,
        filter,
        setFilter,
        addTask,
        editTask,
        toggleTask,
        removeTask,
    };
}

export function AuthForm({ mode }: AuthFormProps) {
    const isRegister = mode === 'register';

    const [values, setValues] = useState<AuthFormValues>({
        email: '',
        password: '',
    });

    const [confirmPassword, setConfirmPassword] = useState('');

    const [errors, setErrors] = useState<AuthFormErrors>({});
    const [serverError, setServerError] = useState<string | null>(null);

    const [submitting, setSubmitting] =
        useState<SubmitMethod | null>(null);

    async function perform(
        operation: () => Promise<void>,
        method: SubmitMethod,
    ): Promise<void> {
        if (submitting) return;

        setServerError(null);
        setSubmitting(method);

        try {
            await operation();
        } catch (error) {
            setServerError(getAuthErrorMessage(error));
        } finally {
            setSubmitting(null);
        }

export function SendSummaryButton() {

                        return result.message;
                    })
                }
            >
                {busy && action === 'summary'
                    ? 'Enviando resumen...'
                    : 'Enviar resumen por email'}
            </button>

            {user && !user.emailVerified && (
                <button
                    className="btn-secondary"
                    type="button"
                    disabled={busy}
                    onClick={() =>
                        void perform('verification', async () => {
                            await requestEmailVerification();

                            return (
                                'Te enviamos un enlace de Firebase. ' +
                                'Abrilo y después volvé a enviar el resumen.'
                            );
                        })
                    }
                >
                    {busy && action === 'verification'
                        ? 'Enviando verificación...'
                        : 'Verificar mi correo'}
                </button>
            )}

            {message && (
                <p
                    className={
                        status === 'error'
                            ? 'alert-error'
                            : 'email-feedback'
                    }
                    role={status === 'error' ? 'alert' : 'status'}
                >
                    {message}
                </p>
            )}
        </div>
    );
}
